    meccha_launcher_core
    STATIC
        src/command_line.cpp
        src/execution.cpp
        src/game_discovery.cpp
        src/loader.cpp
        src/loader_observation.cpp
        src/manifest.cpp
        src/observation_assembly.cpp
        src/ownership.cpp
        src/owned_file_receipt.cpp

        src/workflow.cpp
)

    meccha_launcher_core
    PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
)

    meccha_launcher_core
    PUBLIC
        meccha_build_identity
        MecchaCamouflage::Common
        meccha_core
    PRIVATE
        glaze::glaze
        meccha_build_options
)

        message(

            TO_CMAKE_PATH

            MECCHA_PAYLOAD_CAB_RESOURCE
        )

            resources/launcher_payload.rc.in
            "${CMAKE_CURRENT_BINARY_DIR}/launcher_payload.rc"
